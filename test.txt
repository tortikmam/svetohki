// --- СТРУКТУРЫ ---

struct FlowerBase {
    int id;
    std::string name_ru;
    std::string name_latin;
    std::string family;
    std::string genus;
};

struct FlowerFull : FlowerBase {
    float temp_min, temp_max;
    float ph_min, ph_max;
    int light_level;
    int humidity;
    std::string toxicity;
    std::string user_notes;
    std::string created_at;
};

// Глобальные данные
std::vector<FlowerBase> base_list;
std::vector<PlantData*> sortable_list;
// Карта для быстрого доступа к данным по ID после сортировки
std::map<int, FlowerBase> base_map; 

// --- ФУНКЦИИ ---

void ClearSortable() {
    for (auto p : sortable_list) delete p;
    sortable_list.clear();
}

void SyncData() {
    ClearSortable();
    base_map.clear();
    for (const auto& fb : base_list) {
        // Заполняем карту для таблицы
        base_map[fb.id] = fb;
        // Заполняем список для алгоритмов
        PlantData* p = new PlantData();
        p->name_ru = fb.name_ru;
        p->weight = fb.id; 
        sortable_list.push_back(p);
    }
}

void LoadBaseList(std::vector<FlowerBase>& out_list) {
    try {
        pqxx::connection c("host=127.0.0.1 port=5433 dbname=flowers_db user=myuser password=mypassword");
        pqxx::work txn(c);
        pqxx::result r = txn.exec("SELECT id, name_ru, name_latin, family, genus FROM flowers_base ORDER BY id");
        
        out_list.clear();
        for (auto row : r) {
            out_list.push_back({
                row["id"].as<int>(),
                row["name_ru"].is_null() ? "---" : row["name_ru"].c_str(),
                row["name_latin"].c_str(),
                row["family"].is_null() ? "---" : row["family"].c_str(),
                row["genus"].is_null() ? "---" : row["genus"].c_str()
            });
        }
        SyncData();
    } catch (const std::exception &e) { std::cerr << "DB Error: " << e.what() << std::endl; }
}

FlowerFull GetFullDetails(int id) {
    FlowerFull f;
    try {
        pqxx::connection c("host=127.0.0.1 port=5433 dbname=flowers_db user=myuser password=mypassword");
        pqxx::work txn(c);
        std::string sql = "SELECT b.*, t.temp_min_c, t.temp_max_c, t.ph_min, t.ph_max, t.light_level, t.humidity, t.toxicity, t.user_notes "
                          "FROM flowers_base b LEFT JOIN flowers_technical t ON b.id = t.flower_id WHERE b.id = " + std::to_string(id);
        pqxx::row row = txn.exec1(sql);
        f.id = row["id"].as<int>();
        f.name_ru = row["name_ru"].is_null() ? "---" : row["name_ru"].c_str();
        f.name_latin = row["name_latin"].c_str();
        f.family = row["family"].is_null() ? "---" : row["family"].c_str();
        f.genus = row["genus"].is_null() ? "---" : row["genus"].c_str();
        f.created_at = row["created_at"].c_str();
        f.temp_min = row["temp_min_c"].as<float>(0.0f);
        f.temp_max = row["temp_max_c"].as<float>(0.0f);
        f.ph_min = row["ph_min"].as<float>(0.0f);
        f.ph_max = row["ph_max"].as<float>(0.0f);
        f.light_level = row["light_level"].as<int>(0);
        f.humidity = row["humidity"].as<int>(0);
        f.toxicity = row["toxicity"].is_null() ? "Нет" : row["toxicity"].c_str();
        f.user_notes = row["user_notes"].is_null() ? "" : row["user_notes"].c_str();
    } catch (...) {}
    return f;
}

int main(int argc, char *argv[]) {
    if (SDL_Init(SDL_INIT_VIDEO | SDL_INIT_TIMER) != 0) return -1;
    SDL_Window* window = SDL_CreateWindow("Flower DB", SDL_WINDOWPOS_CENTERED, SDL_WINDOWPOS_CENTERED, 1280, 720, SDL_WINDOW_OPENGL | SDL_WINDOW_RESIZABLE);
    SDL_GLContext gl_context = SDL_GL_CreateContext(window);
    glewInit();

    ImGui::CreateContext();
    ImGuiIO& io = ImGui::GetIO();
    io.ConfigFlags |= ImGuiConfigFlags_DockingEnable;
    io.Fonts->AddFontFromFileTTF("/mnt/c/Windows/Fonts/arial.ttf", 18.0f, NULL, io.Fonts->GetGlyphRangesCyrillic());
    
    ImGui_ImplSDL2_InitForOpenGL(window, gl_context);
    ImGui_ImplOpenGL3_Init("#version 330");

    FlowerFull selected_flower;
    bool show_details = false;
    bool running = true;
    char search_query[128] = "";

    LoadBaseList(base_list);

    while (running) {
        SDL_Event event;
        while (SDL_PollEvent(&event)) {
            ImGui_ImplSDL2_ProcessEvent(&event);
            if (event.type == SDL_QUIT) running = false;
        }

        ImGui_ImplOpenGL3_NewFrame();
        ImGui_ImplSDL2_NewFrame();
        ImGui::NewFrame();
        ImGui::DockSpaceOverViewport(0, nullptr, ImGuiDockNodeFlags_None);

        // --- ОКНО СПИСКА ---
        ImGui::Begin("Реестр растений");
        
        if (ImGui::Button("Обновить данные")) LoadBaseList(base_list);
        ImGui::SameLine();
        if (ImGui::Button("Сортировать по названию")) {
            if (!sortable_list.empty())
                quickSort(sortable_list, 0, sortable_list.size() - 1, cmpByNameRU);
        }

        ImGui::Separator();
        ImGui::InputText("Поиск", search_query, IM_ARRAYSIZE(search_query));
        ImGui::SameLine();
        if (ImGui::Button("Найти")) {
            quickSort(sortable_list, 0, sortable_list.size() - 1, cmpByNameRU);
            int idx = binarySearch(sortable_list, search_query);
            if (idx != -1) {
                selected_flower = GetFullDetails(sortable_list[idx]->weight);
                show_details = true;
            }
        }
        ImGui::Separator();

        // ТАБЛИЦА С 5 СТОЛБЦАМИ
        if (ImGui::BeginTable("BaseTable", 5, ImGuiTableFlags_Borders | ImGuiTableFlags_RowBg | ImGuiTableFlags_Resizable | ImGuiTableFlags_ScrollY)) {
            ImGui::TableSetupColumn("ID", ImGuiTableColumnFlags_WidthFixed, 40.0f);
            ImGui::TableSetupColumn("Название (RU)");
            ImGui::TableSetupColumn("Латынь");
            ImGui::TableSetupColumn("Семейство");
            ImGui::TableSetupColumn("Род");
            ImGui::TableHeadersRow();

            // Итерируемся по sortable_list, чтобы соблюдать порядок сортировки
            for (const auto& p : sortable_list) {
                FlowerBase& fb = base_map[p->weight]; // Достаем полные данные из карты по ID

                ImGui::TableNextRow();
                ImGui::TableSetColumnIndex(0);
                
                char label[32]; sprintf(label, "%d", fb.id);
                if (ImGui::Selectable(label, selected_flower.id == fb.id, ImGuiSelectableFlags_SpanAllColumns)) {
                    selected_flower = GetFullDetails(fb.id);
                    show_details = true;
                }

                ImGui::TableSetColumnIndex(1); ImGui::Text("%s", fb.name_ru.c_str());
                ImGui::TableSetColumnIndex(2); ImGui::Text("%s", fb.name_latin.c_str());
                ImGui::TableSetColumnIndex(3); ImGui::Text("%s", fb.family.c_str());
                ImGui::TableSetColumnIndex(4); ImGui::Text("%s", fb.genus.c_str());
            }
            ImGui::EndTable();
        }
        ImGui::End();

        // --- ОКНО ДЕТАЛЕЙ ---
        if (show_details) {
            ImGui::Begin("Подробная информация", &show_details, ImGuiWindowFlags_AlwaysAutoResize);
            ImGui::TextColored(ImVec4(0.4f, 0.7f, 1.0f, 1.0f), "ОСНОВНЫЕ ДАННЫЕ");
            ImGui::Separator();
            ImGui::Text("ID: %d", selected_flower.id);
            ImGui::Text("Название (RU): %s", selected_flower.name_ru.c_str());
            ImGui::Text("Название (Latin): %s", selected_flower.name_latin.c_str());
            ImGui::Text("Семейство: %s", selected_flower.family.c_str());
            ImGui::Text("Род: %s", selected_flower.genus.c_str());
            
            ImGui::Spacing();
            ImGui::TextColored(ImVec4(1.0f, 0.8f, 0.4f, 1.0f), "ТЕХНИЧЕСКИЕ ХАРАКТЕРИСТИКИ");
            ImGui::Separator();
            ImGui::Text("Температура: %.1f - %.1f C", selected_flower.temp_min, selected_flower.temp_max);
            ImGui::Text("PH: %.1f - %.1f", selected_flower.ph_min, selected_flower.ph_max);
            ImGui::Text("Влажность: %d%%", selected_flower.humidity);
            
            if (ImGui::Button("Закрыть", ImVec2(120, 0))) show_details = false;
            ImGui::End();
        }

        ImGui::Render();
        glViewport(0, 0, (int)io.DisplaySize.x, (int)io.DisplaySize.y);
        glClearColor(0.1f, 0.1f, 0.1f, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT);
        ImGui_ImplOpenGL3_RenderDrawData(ImGui::GetDrawData());
        SDL_GL_SwapWindow(window);
    }

    ClearSortable();
    ImGui_ImplOpenGL3_Shutdown();
    ImGui_ImplSDL2_Shutdown();
    ImGui::DestroyContext();
    SDL_GL_DeleteContext(gl_context);
    SDL_DestroyWindow(window);
    SDL_Quit();
    return 0;
}
